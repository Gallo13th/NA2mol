```python
#!/usr/bin/env python
"""Shared utilities for the RNA2smol ligand-leakage audit.

Standardisation / identity
  primary identity : canonical isomeric SMILES (RDKit MolToSmiles)
  parent identity  : largest fragment, charges neutralised (rdMolStandardize)
  scaffold         : Bemis-Murcko scaffold (no chirality)
  fingerprint      : Morgan radius 2, 2048 bit (ECFP4), no chirality
"""
from __future__ import annotations

import gzip
from pathlib import Path

import numpy as np
from rdkit import Chem, DataStructs, RDLogger
from rdkit.Chem import rdFingerprintGenerator
from rdkit.Chem.Scaffolds import MurckoScaffold
from rdkit.Chem.MolStandardize import rdMolStandardize

RDLogger.DisableLog("rdApp.*")

REPO = Path("github_official/DiffEDMol")
CACHE = Path("/tmp/opencode/leakage/cache")
AUDIT = REPO / "audit"

GEN = rdFingerprintGenerator.GetMorganGenerator(radius=2, fpSize=2048)
POP8 = np.array([bin(i).count("1") for i in range(256)], dtype=np.uint8)

UNCHARGER = rdMolStandardize.Uncharger()
LFC = rdMolStandardize.LargestFragmentChooser()
TAUT = rdMolStandardize.TautomerEnumerator()


def mol_from_smiles(smi):
    try:
        return Chem.MolFromSmiles(smi)
    except Exception:
        return None


def canon(smi):
    m = mol_from_smiles(smi)
    if m is None:
        return None
    try:
        return Chem.MolToSmiles(m)
    except Exception:
        return None


def parent_smiles(smi):
    m = mol_from_smiles(smi)
    if m is None:
        return None
    try:
        frag = rdMolStandardize.FragmentParent(m)
        frag = UNCHARGER.uncharge(frag)
        return Chem.MolToSmiles(frag)
    except Exception:
        return None


def murcko(smi):
    m = mol_from_smiles(smi)
    if m is None:
        return None
    try:
        s = MurckoScaffold.MurckoScaffoldSmiles(mol=m, includeChirality=False)
        return s if s else None
    except Exception:
        return None


def fp_packed(smi):
    m = mol_from_smiles(smi)
    if m is None:
        return None
    try:
        fp = GEN.GetFingerprint(m)
    except Exception:
        return None
    bits = np.frombuffer(fp.ToBitString().encode(), dtype=np.uint8) - 48
    return np.packbits(bits)


def on_bits(smi):
    m = mol_from_smiles(smi)
    if m is None:
        return np.zeros(0, dtype=np.int64)
    fp = GEN.GetFingerprint(m)
    return np.fromiter(fp.GetOnBits(), dtype=np.int64)


def tanimoto_packed(query_packed, lib_packed, lib_pc, q_pc):
    inter = POP8[np.bitwise_and(lib_packed, query_packed)].sum(1).astype(np.int32)
    union = lib_pc + q_pc - inter
    with np.errstate(divide="ignore", invalid="ignore"):
        t = inter / union
    return np.nan_to_num(t)


def build_fps(smiles_list):
    """Build a list of RDKit fingerprints (fast C++ bulk similarity later)."""
    fps = []
    for s in smiles_list:
        m = mol_from_smiles(s)
        if m is not None:
            try:
                fps.append(GEN.GetFingerprint(m))
            except Exception:
                pass
    return fps


def bulk_stats(qfp, fps, thresholds=(0.3, 0.4, 0.5, 0.6, 0.7)):
    """Max Tanimoto + counts >= thresholds of one fingerprint vs a list (C++)."""
    sims = np.asarray(DataStructs.BulkTanimotoSimilarity(qfp, fps))
    if sims.size == 0:
        out = {"max": 0.0, "nn_id": -1}
        for th in thresholds:
            out[f"n_ge_{th}"] = 0
        return out
    j = int(sims.argmax())
    out = {"max": float(sims[j]), "nn_id": j}
    for th in thresholds:
        out[f"n_ge_{th}"] = int((sims >= th).sum())
    return out


class BitIndex:
    """Inverted-index ECFP4 nearest-neighbour search over a large library."""

    def __init__(self, prefix, packed=None, pc=None):
        z = np.load(str(prefix) + ".bitindex.npz")
        self.mol_sorted = z["mol_sorted"]
        self.ptr = z["ptr"]
        self.packed = packed
        self.pc = pc

    @classmethod
    def load(cls, prefix, with_packed=False):
        prefix = Path(prefix)
        z = np.load(str(prefix) + ".bitindex.npz")
        packed = np.load(str(prefix) + ".packed.npy") if with_packed else None
        pc = np.load(str(prefix) + ".popcnt.npy").astype(np.int32)
        self = cls.__new__(cls)
        self.mol_sorted = z["mol_sorted"]
        self.ptr = z["ptr"]
        self.packed = packed
        self.pc = pc
        return self

    @property
    def n(self):
        return len(self.pc)

    def sims(self, bits):
        """Tanimoto similarity of one fingerprint (given its on-bits) to all library rows."""
        if not hasattr(self, "_acc"):
            self._acc = np.zeros(self.n, dtype=np.int64)
        acc = self._acc
        if len(bits) == 0:
            acc.fill(0)
            return acc.astype(np.float64)
        parts = [self.mol_sorted[self.ptr[b]:self.ptr[b + 1]] for b in bits]
        idx = np.concatenate(parts) if parts else np.zeros(0, dtype=np.int32)
        if idx.size == 0:
            acc.fill(0)
            return acc.astype(np.float64)
        acc.fill(0)
        np.add.at(acc, idx, 1)
        qc = float(len(bits))
        with np.errstate(divide="ignore", invalid="ignore"):
            t = acc / (self.pc + qc - acc)
        return np.nan_to_num(t)

    def query_stats(self, bits, thresholds=(0.3, 0.4, 0.5, 0.6, 0.7)):
        t = self.sims(bits)
        j = int(np.argmax(t))
        out = {"max": float(t[j]), "nn_id": j}
        for th in thresholds:
            out[f"n_ge_{th}"] = int((t >= th).sum())
        return out

    def query(self, smi, thresholds=(0.3, 0.4, 0.5, 0.6, 0.7)):
        m = mol_from_smiles(smi)
        if m is None:
            return None
        fp = GEN.GetFingerprint(m)
        bits = np.fromiter(fp.GetOnBits(), dtype=np.int64)
        st = self.query_stats(bits, thresholds)
        st["nn_smiles"] = self.nn_smiles
        return st

    def query(self, smi, thresholds=(0.3, 0.4, 0.5, 0.6, 0.7)):
        """Max similarity + counts >= thresholds for one SMILES."""
        m = mol_from_smiles(smi)
        if m is None:
            return None
        fp = GEN.GetFingerprint(m)
        bits = np.fromiter(fp.GetOnBits(), dtype=np.int64)
        _, t = self.sims_for_bits(bits)
        mx = float(t.max()) if t.size else 0.0
        nn = int(np.argmax(t)) if t.size else -1
        out = {"max": mx}
        for th in thresholds:
            out[f"n_ge_{th}"] = int((t >= th).sum())
        return out, mx, nn, t


def read_smiles_file(path):
```

## B2. `scripts/build_lib.py`

Build a packed ECFP4 fingerprint library from a SMILES file.

```python
#!/usr/bin/env python
"""Build a packed ECFP4 fingerprint library from a SMILES file.

Usage: build_lib.py <smiles_file[.gz]> <out_prefix> [--unique]

Writes:
  <out_prefix>.smiles.txt   one canonical isomeric SMILES per row (order preserved)
  <out_prefix>.packed.npy   uint8 [N,256] packed Morgan r2 2048-bit fingerprints
  <out_prefix>.popcnt.npy   int16 [N] popcounts
  <out_prefix>.fail.txt     rows that failed parse
"""
import sys
import gzip
import time
from multiprocessing import Pool

import numpy as np
from rdkit import Chem, RDLogger
from rdkit.Chem import rdFingerprintGenerator

RDLogger.DisableLog('rdApp.*')
GEN = rdFingerprintGenerator.GetMorganGenerator(radius=2, fpSize=2048)


def canon_fp(smi):
    m = Chem.MolFromSmiles(smi)
    if m is None:
        return None
    try:
        c = Chem.MolToSmiles(m)
        fp = GEN.GetFingerprint(m)
    except Exception:
        return None
    bits = np.frombuffer(fp.ToBitString().encode(), dtype=np.uint8) - 48
    packed = np.packbits(bits)  # 256 bytes
    return c, packed


def main():
    src = sys.argv[1]
    out = sys.argv[2]
    unique = "--unique" in sys.argv
    t0 = time.time()
    opener = gzip.open if src.endswith(".gz") else open
    smis = []
    with opener(src, "rt") as f:
        for line in f:
            s = line.strip()
            if s and s not in ("smiles", "SMILES"):
                smis.append(s)
    print(f"[{time.time()-t0:.0f}s] read {len(smis)} raw smiles", flush=True)
    if unique:
        smis = list(dict.fromkeys(smis))
        print(f"  unique -> {len(smis)}", flush=True)

    with Pool(12) as p:
        res = p.map(canon_fp, smis, chunksize=2000)
    print(f"[{time.time()-t0:.0f}s] parsed", flush=True)

    cans, packs, fails = [], [], []
    seen = set()
    for s, r in zip(smis, res):
        if r is None:
            fails.append(s)
        else:
            if unique and r[0] in seen:
                continue
            seen.add(r[0])
            cans.append(r[0])
            packs.append(r[1])
    print(f"  canonical unique -> {len(cans)}", flush=True)
    packed = np.stack(packs)
    popcnt = np.unpackbits(packed, axis=1).sum(axis=1).astype(np.int16)
    np.save(out + ".packed.npy", packed)
    np.save(out + ".popcnt.npy", popcnt)
    with open(out + ".smiles.txt", "w") as f:
        f.write("\n".join(cans) + "\n")
    with open(out + ".fail.txt", "w") as f:
        f.write("\n".join(fails) + ("\n" if fails else ""))
    print(f"[{time.time()-t0:.0f}s] done N={len(cans)} fails={len(fails)} "
          f"packed={packed.nbytes/1e6:.0f}MB", flush=True)


if __name__ == "__main__":
    main()
```

## B3. `scripts/build_index.py`

Build the inverted bit index over a packed fingerprint library.

```python
#!/usr/bin/env python
"""Build an inverted bit index over ECFP4 fingerprints for fast Tanimoto.

Input : <prefix>.smiles.txt (canonical SMILES, one per row)
Output: <prefix>.bitindex.npz with
          mol_sorted : int32, molecule ids grouped by set bit
          ptr        : int64 [2049], posting-list boundaries per bit
          pc         : int16 [N], popcounts (already available separately)
        <prefix>.bitset.npy  uint8 [N,256] packed bits (already built)
"""
import sys

import numpy as np
from rdkit import Chem, RDLogger
from rdkit.Chem import rdFingerprintGenerator
from multiprocessing import Pool

RDLogger.DisableLog('rdApp.*')
GEN = rdFingerprintGenerator.GetMorganGenerator(radius=2, fpSize=2048)


def onbits(smi):
    m = Chem.MolFromSmiles(smi)
    if m is None:
        return np.zeros(0, dtype=np.int16)
    fp = GEN.GetFingerprint(m)
    return np.fromiter(fp.GetOnBits(), dtype=np.int16)


def main():
    smiles_file = sys.argv[1]
    out = sys.argv[2]
    with open(smiles_file) as f:
        smis = [l.strip() for l in f if l.strip()]
    n = len(smis)
    print("building on-bits for", n, flush=True)
    with Pool(12) as p:
        res = p.map(onbits, smis, chunksize=2000)
    lengths = np.fromiter((len(r) for r in res), dtype=np.int64, count=n)
    flat_bit = np.concatenate(res).astype(np.int32)
    flat_mol = np.repeat(np.arange(n, dtype=np.int32), lengths)
    print("nnz", flat_bit.shape[0], flush=True)
    order = np.argsort(flat_bit, kind="stable")
    bit_sorted = flat_bit[order]
    mol_sorted = flat_mol[order]
    del order, flat_bit, flat_mol
    counts = np.bincount(bit_sorted, minlength=2048)
    ptr = np.zeros(2049, dtype=np.int64)
    ptr[1:] = np.cumsum(counts)
    np.savez(out + ".bitindex.npz", mol_sorted=mol_sorted, ptr=ptr)
    print("saved", out + ".bitindex.npz", flush=True)


if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python
"""Build per-source metadata for the leakage audit.

Sources
  ft_train            fine-tuning ligands, split == train
  ft_train_protein    ft_train, role ft_protein(*)
  ft_train_rna        ft_train, role ft_rna
  ft_val              fine-tuning ligands, split == val
  ft_unused           fine-tuning ligands, split == unused
  ft_dude_dropped     fine-tuning ligands, split == train_dude_dropped
  pretrain            ligand-only pretraining library (canonical unique)

import sys
import time

def build_meta(name, smiles_list, out_dir, pool):
    smiles_list = list(dict.fromkeys(smiles_list))
    print(f"[{name}] {len(smiles_list)} unique smiles", flush=True)
    t = time.time()
    res = pool.map(meta_worker, smiles_list, chunksize=2000)
    parents = {p for p, s in res if p}
    scaffolds = {s for p, s in res if s}
    obj = dict(name=name, n=len(smiles_list), canon=set(smiles_list),
               parent=parents, scaffold=scaffolds)
    with open(out_dir / f"src_{name}.pkl", "wb") as f:

    print(f"[{name}] canonical={len(obj['canon'])} parent={len(parents)} "

    from lka_common import fp_packed
    arr = [fp_packed(s) for s in smiles_list]
    arr = [a for a in arr if a is not None]
    packed = np.stack(arr)
    pc = np.unpackbits(packed, axis=1).sum(1).astype(np.int16)
    np.save(out_dir / f"src_{name}.packed.npy", packed)
    np.save(out_dir / f"src_{name}.popcnt.npy", pc)

if __name__ == "__main__":
```

## B11. `scripts/exp2_summary.py`

```python
#!/usr/bin/env python
"""Combine Experiment IIa (full) and IIb (stratified pretrain sample) into
tables/generated_overlap_summary.csv.

Fine-tuning overlap is exact for all unique generated molecules. Pretraining
similarity fractions are estimates from the stratified sample (with Wilson
95% intervals) because the full 613k x 2.29M sweep exceeds the available
compute; exact-identity pretraining overlap is still computed for all molecules.
"""
import math
import sys
from pathlib import Path

import numpy as np
import pandas as pd

sys.path.insert(0, str(Path(__file__).resolve().parent))
from lka_common import AUDIT, CACHE

THRESHOLDS = (0.3, 0.4, 0.5, 0.6, 0.7)


def wilson(k, n, z=1.96):
    if n == 0:
        return (np.nan, np.nan)
    p = k / n
    d = 1 + z * z / n
    c = (p + z * z / (2 * n)) / d
    h = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / d
    return (max(0.0, c - h), min(1.0, c + h))


def main():
    gen = pd.read_csv(AUDIT / "tables" / "generated_long.csv")
    mol = pd.read_csv(AUDIT / "tables" / "generated_overlap_molecules.csv")
    sample = pd.read_csv(AUDIT / "tables" / "generated_overlap_pretrain_sample.csv")
    g = gen.merge(mol, on="canonical", how="left")
    sample_smi = set(sample["canonical"])
    sm = sample.set_index("canonical")

    rows = []
    for (bench, method), sub in g.groupby(["benchmark", "method"]):
        u = sub.drop_duplicates("canonical")
        rec = dict(benchmark=bench, method=method,
                   n_unique=len(u), n_records=len(sub))
        # fine-tuning: full
        for src in ["ft_train"]:
            rec[f"exact_{src}_uniq"] = int(u[f"exact_{src}"].sum())
            rec[f"exact_{src}_frac"] = round(u[f"exact_{src}"].mean(), 4)
            rec[f"parent_{src}_uniq"] = int(u[f"parent_{src}"].sum())
            rec[f"scaffold_{src}_uniq"] = int(u[f"scaffold_{src}"].sum())
            for th in THRESHOLDS:
                rec[f"ts{th}_{src}_frac"] = round(float((u[f"maxts_{src}"] >= th).mean()), 4)
        # pretraining: exact identity full, similarity from sample
        rec["exact_pretrain_uniq"] = int(u["exact_pretrain"].sum())
        rec["exact_pretrain_frac"] = round(u["exact_pretrain"].mean(), 4)
        rec["parent_pretrain_uniq"] = int(u["parent_pretrain"].sum())
        rec["scaffold_pretrain_uniq"] = int(u["scaffold_pretrain"].sum())
        n_s = int(u["canonical"].isin(sample_smi).sum())
        rec["pretrain_sample_n"] = n_s
        vals = sm.reindex(u.loc[u["canonical"].isin(sample_smi), "canonical"])["maxts_pretrain"].dropna()
        for th in THRESHOLDS:
            if n_s:
                k = int((vals >= th).sum())
                lo, hi = wilson(k, n_s)
                rec[f"ts{th}_pretrain_frac"] = round(k / n_s, 4)
                rec[f"ts{th}_pretrain_lo"] = round(lo, 4)
                rec[f"ts{th}_pretrain_hi"] = round(hi, 4)
```

```python
#!/usr/bin/env python
"""Experiment III: is the published ECFP4-TS recovery robust to overlap?

Reproduces the target-level recovery (mean over tau in [0.25,0.5] of the
fraction of targets with >=1 molecule reaching ECFP4-TS >= tau) and repeats it
after excluding reference actives that overlap the fine-tuning training set or
the pretraining library. All methods are scored on a common eligible-target set.
The permutation null keeps molecule-level similarities and shuffles the
molecule-to-target labels (documented definition).

Outputs
  tables/recovery_original_vs_overlap_filtered.csv
  tables/recovery_curves.csv

import sys
from multiprocessing import Pool
from pathlib import Path

import numpy as np
import pandas as pd

from rdkit.Chem import rdFingerprintGenerator

sys.path.insert(0, str(Path(__file__).resolve().parent))
from lka_common import AUDIT

RDLogger.DisableLog("rdApp.*")

N_PERM = 300

}


def ffp(smi):
    m = Chem.MolFromSmiles(smi)

def keep_mask(ov, filt):
    if filt == "original":
        return np.ones(len(ov), dtype=bool)
    if filt == "ft_exact":
        return ~ov["exact_ft_train"].to_numpy(bool)
    if filt == "pretrain_exact":
        return ~ov["exact_pretrain"].to_numpy(bool)
    if filt.startswith("ft_T"):
        return ov["maxts_ft_train"].to_numpy(float) < float(filt.split("T")[1])
    if filt.startswith("pretrain_T"):
        return ov["maxts_pretrain"].to_numpy(float) < float(filt.split("T")[1])
    raise ValueError(filt)


def target_work(target):
    bench, tid = target["benchmark"], target["target_id"]
    act, ov, gen = target["actives"], target["overlap"], target["gen"]
    afps = [f for f in (ffp(s) for s in act["canonical"]) if f is not None]
    if not afps:
        return None
    uniq = list(dict.fromkeys(gen["canonical"].tolist()))
    gfps = {s: ffp(s) for s in uniq}
    uniq = [s for s in uniq if gfps[s] is not None]
    if not uniq:
        return None
    S = np.zeros((len(uniq), len(afps)), dtype=np.float32)
    for i, s in enumerate(uniq):
        S[i] = DataStructs.BulkTanimotoSimilarity(gfps[s], afps)
    idx = {s: i for i, s in enumerate(uniq)}
    mrows = {m: np.array([idx[s] for s in gen[gen["method"] == m]["canonical"] if s in idx])
             for m in METHODS[bench]}
    out = {}
    for filt in FILTERS:
        keep = keep_mask(ov, filt)
        if keep.sum() == 0 or not all(len(v) for v in mrows.values()):
            out[filt] = None
            continue
        tmax = S[:, keep].max(axis=1)
        out[filt] = {m: tmax[rows].astype(np.float32) for m, rows in mrows.items()}
    return (bench, tid, out)

    if len(sims) == 0:
        return np.nan
    g = pd.Series(sims).groupby(labels).max().to_numpy(float)
    r = (g[None, :] >= TAU[:, None]).mean(axis=1)
    auc = np.trapezoid(r, TAU) if hasattr(np, "trapezoid") else np.trapz(r, TAU)

        return np.trapezoid(y, x, axis=axis) if hasattr(np, "trapezoid") else np.trapz(y, x, axis=axis)

    # analysis (the null requires the full molecule x target matrix; it is not
    # recomputed for the filtered reference sets - see the audit summary).
    pub = {"SMRT-Net": AUDIT.parent / "notebooks" / "output" / "tables" / "smrtbench" / "ecfp4_permutation_gain.csv",

                "ED2Mol": "ED2Mol", "ed2mol": "ED2Mol"}

    for bench, path in pub.items():

            p["method"] = p["method"].map(name_map)
            p["benchmark"] = bench
            frames.append(p)
    if frames:
        pdf = pd.concat(frames)[["benchmark", "method", "ecfp4_A_obs", "ecfp4_A_rand",
                                 "max_ecfp4_tanimoto_raw"]].rename(
            columns={"ecfp4_A_obs": "published_A_obs", "ecfp4_A_rand": "published_A_rand",
                     "max_ecfp4_tanimoto_raw": "published_gain_S"})
        out = out.merge(pdf, on=["benchmark", "method"], how="left")
    out.to_csv(AUDIT / "tables" / "recovery_original_vs_overlap_filtered.csv", index=False)
    pd.DataFrame(curve_rows).to_csv(AUDIT / "tables" / "recovery_curves.csv", index=False)
    print(out.to_string(index=False))

    main()
```

```python
#!/usr/bin/env python
"""Experiment V - ablation-ladder recovery on SMRT-Net.

Adds the no-pretraining baseline (RNA2smol (Baseline), pretraining omitted) and
the intermediate configurations (Pretrain Only, Pretrain+Constrain) to the
recovery-sensitivity and fixed-target analyses, so the contribution of
electron-density pretraining to the ECFP4 recovery (and to its sensitivity to
the reference filters) can be separated from the shared representation.

Configurations (generated_long tags):
  ablation:RNA2smol (Baseline)   no pretraining (lambda_cont = lambda_constrain = 0)
  ablation:Pretrain Only         pretraining, no regularisation
  ablation:Pretrain+Constrain    pretraining + constrained learning
  RNA2smol                       full model (pretraining + constraint + contrastive)

Outputs
  tables/recovery_ablation_ladder.csv
  tables/recovery_ablation_coverage.csv
"""
import sys
from multiprocessing import Pool
from pathlib import Path

import numpy as np
import pandas as pd
from rdkit import Chem, DataStructs, RDLogger
from rdkit.Chem import rdFingerprintGenerator

sys.path.insert(0, str(Path(__file__).resolve().parent))
from lka_common import AUDIT

RDLogger.DisableLog("rdApp.*")
GEN = rdFingerprintGenerator.GetMorganGenerator(radius=2, fpSize=2048)
TAU = np.linspace(0.25, 0.5, 201)
SPAN = 0.25
N_BOOT = 2000
SEED = 0

FILTERS = ["original", "ft_exact", "ft_T0.5", "ft_T0.6", "ft_T0.7",
           "pretrain_exact", "pretrain_T0.5"]
CONFIGS = ["ablation:RNA2smol (Baseline)", "ablation:Pretrain Only",
           "ablation:Pretrain+Constrain", "RNA2smol"]


def ffp(smi):
    m = Chem.MolFromSmiles(smi)
    return GEN.GetFingerprint(m) if m is not None else None


def keep_mask(ov, filt):
    if filt == "original":
        return np.ones(len(ov), dtype=bool)
    if filt == "ft_exact":
        return ~ov["exact_ft_train"].to_numpy(bool)
    if filt == "pretrain_exact":
        return ~ov["exact_pretrain"].to_numpy(bool)
    if filt.startswith("ft_T"):
        return ov["maxts_ft_train"].to_numpy(float) < float(filt.split("T")[1])
    if filt.startswith("pretrain_T"):
        return ov["maxts_pretrain"].to_numpy(float) < float(filt.split("T")[1])
    raise ValueError(filt)


def target_work(target):
    tid = target["target_id"]
    act, ov, gen = target["actives"], target["overlap"], target["gen"]
    afps = [f for f in (ffp(s) for s in act["canonical"]) if f is not None]
    if not afps:
        return None
    uniq = list(dict.fromkeys(gen["canonical"].tolist()))
    gfps = {s: ffp(s) for s in uniq}
    uniq = [s for s in uniq if gfps[s] is not None]
    if not uniq:
        return None
    S = np.zeros((len(uniq), len(afps)), dtype=np.float32)
    for i, s in enumerate(uniq):
        S[i] = DataStructs.BulkTanimotoSimilarity(gfps[s], afps)
    idx = {s: i for i, s in enumerate(uniq)}
    out = {}
    for m in CONFIGS:
        rows = np.array([idx[s] for s in gen[gen["method"] == m]["canonical"] if s in idx])
        if len(rows) == 0:
            continue
        d = {"original": S[rows, :].max(axis=1).astype(np.float32)}
        for filt in FILTERS[1:]:
            keep = keep_mask(ov, filt)
            d[filt] = (S[rows][:, keep].max(axis=1).astype(np.float32) if keep.sum() else None)
        out[m] = d
    return (tid, out)


def trapz(y, x, axis=-1):
    return np.trapezoid(y, x, axis=axis) if hasattr(np, "trapezoid") else np.trapz(y, x, axis=axis)


def curve(g):
    g = np.asarray(g, float)
    g = g[np.isfinite(g)]
    if len(g) == 0:
        return np.nan
    r = (g[:, None] >= TAU[None, :]).mean(axis=0)
    return float(trapz(r, TAU) / SPAN)


def boot_ci(g, rng):
    g = np.asarray(g, float)
    g = g[np.isfinite(g)]
    if len(g) == 0:
        return (np.nan, np.nan)
    G = (g[:, None] >= TAU[None, :])
    bidx = rng.integers(0, len(g), size=(N_BOOT, len(g)))
    b = trapz(G[bidx].mean(axis=1), TAU, axis=1) / SPAN
    return float(np.percentile(b, 2.5)), float(np.percentile(b, 97.5))


def main():
    ref = pd.read_csv(AUDIT / "tables" / "reference_actives.csv")
    ref = ref[ref.parse_ok]
    ov = pd.read_csv(AUDIT / "tables" / "reference_active_overlap.csv", low_memory=False)
    gen = pd.read_csv(AUDIT / "tables" / "generated_long.csv")
    gen = gen[gen.benchmark == "SMRT-Net"]
    ref = ref[ref.benchmark == "SMRT-Net"]
    ref["_k"] = ref["benchmark"] + "|" + ref["target_id"].astype(str) + "|" + ref["canonical"]
    ov["_k"] = ov["benchmark"] + "|" + ov["target_id"].astype(str) + "|" + ov["canonical"]
    key = ["exact_ft_train", "exact_pretrain", "maxts_ft_train", "maxts_pretrain"]
    ref = ref.merge(ov[key + ["_k"]].drop_duplicates("_k"), on="_k", how="left")

    targets = []
    for tid, act in ref.groupby("target_id"):
        g = gen[gen.target_id == tid]
        if len(g) and len(act):
            targets.append(dict(target_id=tid, actives=act,
                                overlap=act.reset_index(drop=True), gen=g))
    print("SMRT-Net targets", len(targets), flush=True)
    with Pool(12) as p:
        res = [r for r in p.map(target_work, targets, chunksize=2) if r]

    data = {}   # config -> filt -> {tid: g}
    for tid, out in res:
        for m, d in out.items():
            for filt, arr in d.items():
                if arr is not None:
                    data.setdefault(m, {}).setdefault(filt, {})[tid] = float(np.nanmax(arr))

    # coverage
    cov = [dict(config=m, targets_covered=len(data.get(m, {}).get("original", {})))
           for m in CONFIGS]
    cover = set.intersection(*[set(data.get(m, {}).get("original", {})) for m in CONFIGS])
    cov.append(dict(config="COMMON", targets_covered=len(cover)))
    pd.DataFrame(cov).to_csv(AUDIT / "tables" / "recovery_ablation_coverage.csv", index=False)

    rng = np.random.default_rng(SEED)
    rows = []
    for filt in FILTERS:
        elig = None
        for m in CONFIGS:
            s = set(data.get(m, {}).get(filt, {}).keys())
            elig = s if elig is None else (elig & s)
        elig = sorted(elig or set())
        for m in CONFIGS:
            gf = np.array([data[m][filt][t] for t in elig])
            go = np.array([data[m]["original"][t] for t in elig])
            a_f = curve(gf)
            a_o = curve(go)
            lo, hi = boot_ci(gf, rng)
            rows.append(dict(filter=filt, config=m, n_eligible=len(elig),
                             A_original_subset=round(a_o, 4),
                             A_filtered_subset=round(a_f, 4),
                             delta=round(a_f - a_o, 4),
                             A_filtered_boot_lo=round(lo, 4),
                             A_filtered_boot_hi=round(hi, 4)))
    out = pd.DataFrame(rows)
    out.to_csv(AUDIT / "tables" / "recovery_ablation_ladder.csv", index=False)
    print("\n=== SMRT-Net ablation ladder: A_obs(ECFP4) per filter ===")
    piv = out.pivot_table(index="filter", columns="config", values="A_filtered_subset")
    print(piv.to_string())
    print("\n=== same-target original vs filtered (delta) ===")
    print(out.pivot_table(index="filter", columns="config", values="delta").to_string())


if __name__ == "__main__":
    main()
```

# Appendix - audit summary

# Ligand-leakage audit — methods, findings and recommended wording

Scope: quantify chemical overlap between the SMRT-Net and DUD-E evaluation
references and (i) the fine-tuning training ligands, (ii) the ligand-only
pretraining library, and (iii) the generated molecules; and test whether the
manuscript's target-level ECFP4-Tanimoto recovery result is robust to that
overlap.

## 1. Data and methods

| Item | Choice |
|------|--------|
| Fine-tuning ligands | `finetuning_ligands_table.tar.gz` → `training_ligands_leakage_table.csv` (225,239 records; splits `train` 127,255 / `val` 27,621 / `unused` 67,536 / `train_dude_dropped` 2,827; roles protein(crossdock/expdb) and RNA). No explicit test split exists; `unused` is reported separately and is **not** counted as gradient-update exposure. |
| Pretraining library | `pretrain_smiles.txt.gz` → 2,287,199 unique canonical SMILES (ligand-only, no pocket labels). |
| SMRT-Net references | `notebooks/RNALigBench/activesmi/{pdb}_pos.smi` (fallback `{pdb}.smi`) for the 27 evaluated systems → 96 records, 50 unique actives. |
| DUD-E references | `release_package_v1/notebooks/DUDE_active_compounds.csv` (`label == active`) for the 101 evaluated targets → 35,237 unique actives (one per `(pdb, smiles)`). |
| Generated molecules | `notebooks/output/smrtbench_combined_results.csv` and `notebooks/dude_stat_all.csv` (post-evaluation stage) → 677,552 unique (benchmark, method, target, canonical) records; 613,286 unique canonical molecules. |
| Standardisation | RDKit 2024.09.6. Primary identity = canonical isomeric SMILES; parent identity = largest fragment + uncharge; scaffold = Bemis–Murcko (no chirality). Parsing failures counted (0). |
| Fingerprint | Morgan radius 2, 2048 bits, no chirality (identical to the manuscript). Similarity = Tanimoto. |
| Thresholds | 0.3 / 0.4 / 0.5 / 0.6 / 0.7 (0.5 for continuity with prior work; not a leakage boundary). |
| Recovery statistic | Target-level: for each method, R(τ)=fraction of targets with ≥1 molecule reaching ECFP4-TS ≥ τ; A = mean R over τ∈[0.25,0.5] (the manuscript definition). Bootstrap CIs resample **targets** (2,000 draws). |

Fine-tuning similarity was computed for the full set of reference actives (all
sources, RDKit C++ bulk) and for all 613,286 unique generated molecules
(vs `train`). Pretraining similarity for the **generated** molecules was computed
on a stratified random sample (4,000 per benchmark×method; 67,664 molecules
total) because the full 613k × 2.29M exact sweep is not tractable on the
available 6-physical-core machine; exact-identity pretraining overlap and the
reference-active pretraining similarity were computed for **all** records. The
three file-based deliverables below record what is exact and what is sampled.

## 2. Experiment I — benchmark-active overlap (exact for all actives)

Ligand-level (unique actives); percentages.

**DUD-E (35,237 actives / 101 targets)**

| Source | exact | parent | scaffold | max-TS ≥0.5 | ≥0.7 |
|--------|------:|-------:|---------:|------------:|-----:|

| ft train–protein | 2.16% | 2.70% | 12.25% | 36.4% | 10.9% |
| ft train–RNA | 0.01% | 0.01% | 0.95% | 0.3% | 0.0% |
| ft val | 0.41% | 0.57% | 6.12% | 11.9% | 2.3% |
| ft unused | 0.56% | 1.33% | 7.24% | 18.1% | 3.8% |
| pretrain | 9.19% | 19.09% | 47.61% | 90.0% | 56.6% |

Target-level: 99/101 (98.0%) DUD-E targets have ≥1 active exactly matching a
training ligand and 101/101 (100%) share a scaffold; 94/101 (93.1%) have an

**SMRT-Net (50 actives / 27 targets)**

| Source | exact | parent | scaffold | max-TS ≥0.5 | ≥0.7 |
|--------|------:|-------:|---------:|------------:|-----:|

Target-level: 11/27 (40.7%) SMRT targets have ≥1 exact training match; 2/27

Headline: exact supervised-training duplication is low for DUD-E (2.2% of
actives) but the fraction with a **close** training analogue is substantial
(36% ≥0.5). Because DUD-E targets carry hundreds of actives each, almost every
target still touches the training set at ≥1 active (98%), even though the
per-active rate is small — the two summaries answer different questions.

## 3. Experiment II — generated-output overlap

Fine-tuning overlap is exact for all 613,286 unique generated molecules;
pretraining similarity is from the stratified sample (Wilson 95% CI in
`generated_overlap_summary.csv`).

Main methods (unique molecules):

| Benchmark | Method | exact ft | ≥0.5 ft | exact pre | ≥0.5 pre (95% CI) |
|-----------|--------|---------:|--------:|----------:|------------------:|
| DUD-E | RNA2smol | 0.08% | 7.3% | 0.36% | 26.5% (25.2–27.9) |
| DUD-E | MolCRAFT | 0.06% | 1.0% | 0.04% | 4.0% (3.4–4.6) |
| DUD-E | Pocket2Mol | 0.76% | 8.6% | 0.58% | 23.5% (22.3–24.8) |
| DUD-E | ED2Mol | 0.05% | 4.5% | 0.27% | 22.2% (21.1–23.4) |
| SMRT-Net | RNA2smol | 0.16% | 6.0% | 0.44% | 16.8% (15.8–17.9) |
| SMRT-Net | MolCRAFT | 0.65% | 5.8% | 0.22% | 8.3% (7.5–9.2) |
| SMRT-Net | Pocket2Mol | 0.55% | 10.6% | 0.44% | 25.1% (23.8–26.4) |
| SMRT-Net | ED2Mol | 0.08% | 7.4% | 0.46% | 26.7% (25.4–28.0) |

- RNA2smol reproduces fine-tuning ligands at very low rates (0.08–0.16% exact;
  6–7% within 0.5) — no evidence of supervised memorisation by the main model.
- RNA2smol generated molecules are chemically familiar to the **pretraining**
  library (26.5% within 0.5 for DUD-E) at a level comparable to Pocket2Mol/ED2Mol
  and well above MolCRAFT. This is expected for a model pretrained on 2.29M
  drug-like structures and, since pretraining carries no pocket labels, is not
  direct target-associated leakage.
- The `RNA2smol(AR)` ablation is an outlier (10.1% exact / 55.4% ≥0.5 vs
  fine-tuning training) and is **not** the reported model.

## 4. Experiment III — recovery robustness (`recovery_original_vs_overlap_filtered.csv`)

All methods scored on a common eligible-target set; A = mean target recovery

## 6. Experiment V — ablation ladder: does the advantage depend on pretraining?

`recovery_ablation_ladder.csv`, `recovery_ablation_coverage.csv` (script
`exp5_ablation_recovery.py`). SMRT-Net only; all four configurations cover all
27 targets.

The manuscript's ablation ladder is added to the recovery analysis. The
no-pretraining configuration (`RNA2smol (Baseline)`, pretraining omitted,
λ_cont = λ_constrain = 0) is the key control for the pretraining question.

A_obs at each reference filter (all configs on their common eligible target set):

| filter | full model | Pretrain+Constrain | no-pretrain Baseline | Pretrain Only |
|--------|-----------:|-------------------:|---------------------:|--------------:|
| original (27) | 0.218 | 0.132 | 0.141 | 0.034 |
| ft exact (24) | 0.245 | 0.146 | 0.156 | 0.039 |
| ft analogues ≥0.5 (18) | 0.249 | 0.182 | 0.134 | 0.034 |
| ft analogues ≥0.7 (19) | 0.309 | 0.184 | 0.197 | 0.049 |
| pretrain exact (26) | 0.226 | 0.137 | 0.146 | 0.036 |
| pretrain analogues ≥0.5 (14) | 0.052 | 0.104 | 0.114 | 0.052 |

Same-target filter effect (A_filtered − A_original on the surviving targets):

| filter | full model | Pretrain+Constrain | no-pretrain Baseline |
|--------|-----------:|-------------------:|---------------------:|
| ft analogues ≥0.5 | −0.077 | −0.016 | −0.077 |
| pretrain exact | 0.000 | −0.003 | −0.003 |
| pretrain analogues ≥0.5 | −0.198 | −0.099 | −0.077 |

Findings:

- **The no-pretraining baseline already beats every external generator**
  (SMRT-Net A_obs 0.141 vs MolCRAFT 0.060, Pocket2Mol 0.081, ED2Mol 0.078), so
  part of RNA2smol's recovery advantage comes from the shared representation and
  fine-tuning strategy rather than pretraining.
- **Pretraining alone is not sufficient**: Pretrain Only is the weakest
  configuration (0.034), consistent with its documented geometry pathology; the
  full model's advantage emerges from pretraining combined with constrained and
  contrastive learning.
- **The full model's recovery is markedly more pretraining-dependent**: on the
  same 14 targets, removing actives with a pretraining neighbour ≥0.5 drops the
  full model 0.251→0.052 (Δ −0.198) versus Δ −0.077 for the no-pretraining
  baseline. This quantifies a real contribution of pretraining-adjacent chemistry
  to the headline recovery while showing the representation alone still recovers.
  Note the full model drops *below* the no-pretraining baseline on this filtered
  set (0.052 vs 0.114) — the residual advantage reverts to the no-pretraining
  level once pretraining-adjacent references are removed.

## 7. Interpretation (per the plan's decision framework)

This is closest to **Outcome 2**: overlap with the ligand-only pretraining
library is substantial and is the largest source of chemical familiarity, while
direct exact overlap with supervised training is limited. The headline
target-recovery advantage of RNA2smol over the baselines is **not** explained by
exact duplication of evaluation actives in supervised training. On fixed target
subsets, removing exact training matches leaves recovery unchanged and removing
close training analogues causes only a modest reduction while preserving the
RNA2smol ordering over baselines; the larger pretraining-analogue effect reflects
the sensitivity of the metric to reference chemistry rather than target-specific
leakage. We cannot claim this excludes every form of memorisation.

The ablation ladder sharpens the conclusion: the no-pretraining baseline already
exceeds all external generators, so part of the advantage is attributable to the
shared representation and fine-tuning; but the full model's ECFP4 recovery is
substantially more dependent on pretraining-adjacent reference chemistry
(Δ −0.198 on the pretraining-analogue filter versus −0.077 for the
no-pretraining baseline), and falls to the no-pretraining level once those
references are removed. Ligand-only pretraining therefore makes a real
contribution to the retrospective ECFP4 recovery, without target-pocket labels
and without being the sole source of the advantage.

## 8. Recommended manuscript text (Methods / Limitations)

> To assess potential chemical overlap between the evaluation references and the
> training data, SMRT-Net and DUD-E active ligands were standardised (RDKit
> 2024.09.6; canonical isomeric SMILES, largest-fragment parent) and compared
> with the fine-tuning partitions and the ligand-only pretraining library using
> exact identity, Bemis–Murcko scaffold and maximum ECFP4-Tanimoto similarity
> (Morgan r=2, 2048 bits). Fine-tuning overlap was reported separately from
> pretraining overlap, since pretraining ligands carry no pocket labels. Exact
> matches to the fine-tuning **training** split affected 2.2% of DUD-E and 18%
> of SMRT-Net actives, but close analogues (ECFP4-TS ≥0.5) were more common (36%
> and 58%). Generated molecules were compared with the same libraries: exact
> reproduction of training ligands was rare for RNA2smol (0.08–0.16%), while
> ~17–27% of generated molecules had a pretraining neighbour ≥0.5. Target-level
> ECFP4-TS recovery was reproduced and recomputed after excluding reference
> actives with exact or close supervised-training matches on a common eligible
> target set, with target-level bootstrap resampling for uncertainty. Recovery
> was unchanged by exact-match exclusion and remained above all baselines after
> excluding close training analogues under every threshold tested.

## 10. Deviations and limitations

1. **Pretraining similarity for generated molecules is sampled** (4,000/method;
   67,664 total) because the full 613k×2.29M sweep exceeds the available compute
   (6 physical cores, 15 GB). Exact-identity overlays and fine-tuning
   similarities are exact for all molecules.
2. The permutation-standardised null was not recomputed for the filtered
   reference sets (it requires the full molecule×target similarity matrix);
   published gains are attached for the original analysis, and the filtered
   comparisons rely on target-bootstrap CIs of A(τ).
3. No explicit `test` split exists in the fine-tuning table; `val` is reported
   separately and `unused` is not treated as gradient-update exposure.
4. Only the post-evaluation molecule stage was available; raw generated SDFs
   were not re-scored, so stage-wise attrition of chemical familiarity could not
   be quantified.
5. Target-level coverage differs by benchmark and method. Exp III uses the
   common eligible-target set for within-filter method comparisons; Exp IV adds
   the fixed-target original-vs-filtered comparison to separate reference
   chemistry from target composition. The audit's DUD-E reference-active file
   covers 101 targets vs the manuscript's 102, and per-method molecule coverage
   differs by 1–3 targets, so per-method reconstruction agrees with the
   manuscript's own per-threshold tables to within ≈0.02–0.025 rather than
   exactly.
